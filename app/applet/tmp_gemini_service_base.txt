import { GoogleGenAI, Type } from "@google/genai";
import { ActivityAnalysis, ComparisonAnalysis, CurrentWeatherData, DailyForecastItem, HourlyForecastItem, LocationInfo, TravelAnalysis, WeatherAdvisory, WeatherAnalysis } from "../types.js";
import { getLanguageConfig } from "../../src/languages.js";
import { validateAndGroundWeatherResponse, checkLocationConsistency, StructuredAIOutput } from "./weatherGroundingValidator.js";
import { buildMultilingualDeterministicFallback } from "./multilingualFallback.js";

let genAiInstance: GoogleGenAI | null = null;

function getGenAI(): GoogleGenAI | null {
  if (genAiInstance) return genAiInstance;
  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) {
    return null;
  }
  try {
    genAiInstance = new GoogleGenAI({
      apiKey,
      httpOptions: {
        headers: {
          "User-Agent": "aistudio-build",
        },
      },
    });
    return genAiInstance;
  } catch (err) {
    console.error("Failed to initialize GoogleGenAI:", err);
    return null;
  }
}

export interface GroundedWeatherContext {
  userQuery: string;
  language: string;
  location?: LocationInfo;
  current?: CurrentWeatherData;
  hourly?: HourlyForecastItem[];
  daily?: DailyForecastItem[];
  analysis?: WeatherAnalysis;
  advisories?: WeatherAdvisory[];
  activity?: ActivityAnalysis;
  comparison?: ComparisonAnalysis;
  travel?: TravelAnalysis;
  failedLocations?: string[];
}

function isTransientError(err: any): boolean {
  if (!err) return false;
  const status = Number(err.status || err.code || err.statusCode);
  if (status === 503 || status === 429 || status === 500 || status === 502 || status === 504) {
    return true;
  }
  const msg = (err.message || "").toLowerCase();
  return (
    msg.includes("503") ||
    msg.includes("high demand") ||
    msg.includes("unavailable") ||
    msg.includes("temporary") ||
    msg.includes("spikes in demand") ||
    msg.includes("resource_exhausted") ||
    msg.includes("rate limit") ||
    msg.includes("overloaded") ||
    msg.includes("econnreset") ||
    msg.includes("etimedout")
  );
}

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

export async function generateWeatherExplanation(context: GroundedWeatherContext): Promise<{ reply: string; isAIGenerated: boolean; fallbackUsed: boolean }> {
  const ai = getGenAI();
  const configuredModel = process.env.GEMINI_MODEL || "gemini-3.8-flash";
  const langConfig = getLanguageConfig(context.language);

  // If no API key or AI client is available, generate rich deterministic response
  if (!ai) {
    const rawFallback = generateDeterministicFallback(context);
    const validated = validateAndGroundWeatherResponse(rawFallback, context);
    return {
      reply: validated.correctedText,
      isAIGenerated: false,
      fallbackUsed: true,
    };
  }

  // Prioritized candidate models list to gracefully absorb transient 503 high-demand spikes
  const candidateModels = Array.from(
    new Set([
      configuredModel,
      "gemini-3.8-flash",
      "gemini-flash-latest",
      "gemini-3.1-flash-lite",
    ])
  );

  const systemInstruction = `You are WeatherGPT, a world-class conversational AI weather intelligence engine built for India and global users.
Weather provider data is ALWAYS the absolute source of truth. You are strictly responsible for interpretation, natural language communication, and safe, human-centric guidance.

MANDATORY FACTUAL GROUNDING RULES (NEVER VIOLATE):
1. STRICT LOCATION BINDING (CRITICAL):
   - For single-location queries: You MUST answer exclusively for the target location specified in the context. You MUST NOT hallucinate, adopt, or mention unrelated locations.
   - For comparison queries: You MUST answer for and compare ALL verified locations provided in the comparison data. NEVER state that verified weather data is available exclusively for only one location when data for multiple locations is provided.
   - If a specific location could not be resolved or fetched (indicated in the unresolved locations list), clearly state which location failed instead of silently answering for only one.
2. NEVER modify, round, exaggerate, estimate, or replace numerical weather values received from the weather provider.
3. If the weather provider reports:
   Rain probability = 96%
   You must state exactly 96%, NOT 100%, and NOT "guaranteed".
4. All numerical values must remain identical to the source data, including:
   - Temperature (keep exact source numbers)
   - Feels-like temperature
   - Precipitation probability (%)
   - Precipitation amount (mm)
   - Wind speed and gusts (km/h)
   - Humidity (%) - if null/unavailable, state that humidity data is unavailable, NEVER report 0%
   - UV index
   - Visibility (km)
   - Pressure (hPa)
   - Dates and times
5. ACTIVE RAIN VS FORECAST (CRITICAL):
   - A high precipitation probability does NOT mean rain is currently falling!
   - If current rain is 0 mm/hr, you MUST NOT state that it is currently raining, downpouring, or that active rain is underway.
   - Explicitly clarify whether rain is active right now or if conditions are currently dry with upcoming rain chances.
6. TRAFFIC, ROAD CONDITIONS & DATA-TRUTH CONSTRAINTS (MANDATORY):
   - You do NOT have access to live road condition sensors, road grip telemetry, traffic cameras, or road closure databases.
   - You MUST NOT make claims about actual road conditions such as:
     * roads are dry / wet
     * wet road conditions / wet pavement / wet asphalt
     * road grip / tire traction degradation
     * road visibility / pavement visibility
     * waterlogging / flooded roads
     * traffic congestion / road closures / transit delays
   - You may state that rain or thunderstorms MAY affect travel conditions in general, but must NEVER present actual road conditions as known facts without a real road-condition/traffic data source.
   - Use weather-only wording such as:
     "Rain and thunderstorms are likely, so allow extra travel time and consider carrying rain protection."
   - Do NOT claim "safest" unless supported by the available data. Prefer:
     "weather-wise more suitable" or "better weather conditions for outdoor activity."
7. PROBABILITY VS CERTAINTY:
   - Do NOT convert probability values into absolute statements unless the actual provider value is explicitly 100%.
   - E.g., 96% is a very high probability, but it is NOT 100% and NOT "guaranteed".
   - 70% must NEVER become 80% or 100%.
   - 0% must NEVER become a positive probability.
8. HUMAN-CENTRIC RECOMMENDATIONS:
   - Connect weather data directly to practical daily decisions: clothing, umbrella necessity, outdoor suitability, UV protection.
   - Structure responses with: (1) Direct Answer, (2) Atmospheric Interpretation, (3) Practical Recommendation.

TARGET LANGUAGE SPECIFICATIONS (CRITICAL):
- Selected Authoritative Language: ${langConfig.name} (${langConfig.nativeName}) [Code: ${langConfig.code}, Script: ${langConfig.script}]
- Directive: ${langConfig.geminiPromptInstruction}
- ALL generated text in every field (directAnswer, detailedExplanation, travelWeatherAdvice) MUST be purely and exclusively in ${langConfig.name} (${langConfig.nativeName}).
- DO NOT use English headings, English prefixes, or English disclaimers when the target language is not English.
- For Hinglish: Use natural, conversational Roman-script Hindi mixed with English terms (e.g. "Aaj barish ke asaar 96% hain, halki umbrella carry karein"). Do NOT write in Devanagari script.
- For Indian & International languages: Write fluently and idiomatically in the specified native script (${langConfig.script}), but ALWAYS keep numerical weather values intact and accurate.`;

  const payloadSummary = formatContextForGemini(context);
  const promptContents = `User Query: "${context.userQuery}"\nSelected Language: ${langConfig.name} (${langConfig.code})\n\nVERIFIED WEATHER PROVIDER DATA:\n${payloadSummary}\n\nRespond as WeatherGPT in ${langConfig.name} (${langConfig.nativeName}) maintaining 100% numerical fidelity to the verified provider data:`;

  const generationConfig = {
    systemInstruction,
    temperature: 0.2, // Low temperature for maximum factual fidelity
    responseMimeType: "application/json",
    responseSchema: {
      type: Type.OBJECT,
      properties: {
        directAnswer: {
          type: Type.STRING,
          description: `Direct answer to user query in ${langConfig.name} stating exact factual figures without rounding.`,
        },
        referencedPrecipitationProbability: {
          type: Type.NUMBER,
          description: "The exact precipitation probability from the provider data (e.g. 96). Must NOT be altered.",
        },
        referencedTemperature: {
          type: Type.NUMBER,
          description: "The exact current temperature in Celsius from the provider data.",
        },
        referencedFeelsLike: {
          type: Type.NUMBER,
          description: "The exact feels-like temperature in Celsius from the provider data.",
        },
        referencedWindSpeed: {
          type: Type.NUMBER,
          description: "The exact wind speed in km/h from the provider data.",
        },
        referencedHumidity: {
          type: Type.NUMBER,
          description: "The exact humidity percentage from provider data, or null if unavailable.",
        },
        detailedExplanation: {
          type: Type.STRING,
          description: `Clear conversational explanation and human-centric practical recommendations in ${langConfig.name}.`,
        },
        travelWeatherAdvice: {
          type: Type.STRING,
          description: `If query is travel-related, route weather risk in ${langConfig.name}. DO NOT claim live traffic jams.`,
        },
      },
      required: ["directAnswer", "detailedExplanation"],
    },
  };

  for (const modelToTry of candidateModels) {
    for (let attempt = 0; attempt < 2; attempt++) {
      try {
        const response = await ai.models.generateContent({
          model: modelToTry,
          contents: promptContents,
          config: generationConfig,
        });

        let structured: StructuredAIOutput | undefined;
        let fullText = "";
        const rawText = response.text?.trim() || "";

        try {
          const jsonMatch = rawText.match(/\{[\s\S]*\}/);
          if (jsonMatch) {
            structured = JSON.parse(jsonMatch[0]);
            const parts = [
              structured?.directAnswer,
              structured?.detailedExplanation,
              structured?.travelWeatherAdvice ? `\n\n${structured.travelWeatherAdvice}` : "",
            ].filter(Boolean);
            fullText = parts.join("\n\n");
          } else {
            fullText = rawText;
          }
        } catch {
          fullText = rawText;
        }

        if (!fullText) {
          throw new Error(`Empty response from Gemini model ${modelToTry}`);
        }

        // Check location consistency (prevent foreign hallucinations like Denmark)
        const locConsistency = checkLocationConsistency(fullText, context.location);
        if (!locConsistency.isConsistent) {
          console.warn("Location consistency check failed:", locConsistency.mismatchWarning);
          const rawFallback = generateDeterministicFallback(context);
          const validated = validateAndGroundWeatherResponse(rawFallback, context);
          return {
            reply: validated.correctedText,
            isAIGenerated: false,
            fallbackUsed: true,
          };
        }

        // Pass through Server-Side Validation & Grounding Engine
        const validated = validateAndGroundWeatherResponse(fullText, context, structured);
        if (!validated.isValid) {
          console.warn("Weather grounding violations corrected:", validated.violationsDetected);
        }

        return {
          reply: validated.correctedText,
          isAIGenerated: true,
          fallbackUsed: false,
        };
      } catch (err: any) {
        const transient = isTransientError(err);
        if (transient && attempt === 0) {
          // Brief pause before retry on high demand
          await sleep(350);
          continue;
        }
        // If second attempt or not transient, cascade to next candidate model
        break;
      }
    }
  }

  // If all model candidates failed or experienced demand spikes, cleanly serve deterministic fallback
  console.info("Gemini API models currently unavailable or high demand encountered; serving verified deterministic weather response.");
  const rawFallback = generateDeterministicFallback(context);
  const validated = validateAndGroundWeatherResponse(rawFallback, context);
  return {
    reply: validated.correctedText,
    isAIGenerated: false,
    fallbackUsed: true,
  };
}

function formatContextForGemini(context: GroundedWeatherContext): string {
  const parts: string[] = [];

  parts.push("=== CANONICAL IMMUTABLE WEATHER PROVIDER DATA (GROUND TRUTH) ===");

  if (context.comparison) {
    const comp = context.comparison;
    parts.push(`AUTHORITATIVE COMPARED LOCATIONS:
${comp.locations.map((l) => `- ${l.location.name} (${l.location.admin1 || ""}, ${l.location.country})`).join("\n")}
CRITICAL MANDATE: You are comparing weather between ${comp.locations.map((l) => l.location.name).join(" and ")}.
You MUST compare both locations using the verified weather metrics provided below: temperature high/low, precipitation probability, rain, thunderstorms, and wind.
Never claim that verified weather data is available exclusively for only one location when data for both is provided.
Provide a clear comparison and weather-based recommendation using strictly weather-only wording.`);

    parts.push(`COMPARISON TARGET: ${comp.targetDay ? comp.targetDay.toUpperCase() : "CURRENT"}`);
    parts.push(`COMPARATIVE SUMMARY: ${comp.comparativeSummary}`);
    if (comp.recommendation) {
      parts.push(`WEATHER RECOMMENDATION: ${comp.recommendation}`);
    }
    parts.push(
      `METRIC DELTAS & HIGHLIGHTS:\n${comp.metricDeltas
        .map((m) => `• ${m.metric}: ${m.highlight} (Values: ${m.values.map((v) => `${v.locationName}: ${v.value}`).join(", ")})`)
        .join("\n")}`
    );

    parts.push(
      `DETAILED PER-LOCATION VERIFIED DATA:\n${comp.locations
        .map((locRes) => {
          const loc = locRes.location;
          if (locRes.targetForecast) {
            const tf = locRes.targetForecast;
            return `[${loc.name} - ${tf.targetDay.toUpperCase()} FORECAST]
- High Temperature: ${tf.tempMax}°C
- Low Temperature: ${tf.tempMin}°C
- Condition: ${tf.condition}
- Precipitation Probability: ${tf.precipitationProbability}%
- Rain Volume: ${tf.rainSum} mm
- Thunderstorm Forecast: ${tf.isThunderstorm ? "YES (Thunderstorm forecast)" : "NO (No convective thunderstorm activity forecast)"}
- Wind Speed Max: ${tf.windSpeedMax} km/h
- UV Index Max: ${tf.uvIndexMax}`;
          }
          return `[${loc.name} - CURRENT OBSERVATIONS]
- Temperature: ${locRes.current.temperature}°C (Feels like: ${locRes.current.feelsLike}°C)
- Condition: ${locRes.current.condition}
- Precipitation Probability: ${locRes.current.precipitationProbability}%
- Rain Amount: ${locRes.current.rain} mm/hr
- Wind Speed: ${locRes.current.windSpeed} km/h
- Humidity: ${locRes.current.humidity ?? "N/A"}%`;
        })
        .join("\n\n")}`
    );
  } else if (context.location) {
    parts.push(`AUTHORITATIVE TARGET LOCATION (CRITICAL: ANSWER ONLY FOR THIS LOCATION):
- Location Name: ${context.location.name}
- Administrative Region: ${context.location.admin1 || "N/A"}
- Country: ${context.location.country || "N/A"} (Country Code: ${context.location.countryCode || "N/A"})
- Coordinates: ${context.location.latitude.toFixed(4)}°N, ${context.location.longitude.toFixed(4)}°E
- Timezone: ${context.location.timezone}
CRITICAL MANDATE: You are answering strictly and exclusively for ${context.location.name} (${context.location.country}). You MUST NOT mention, borrow from, or confuse this with any other location.`);
  }

  if (context.failedLocations && context.failedLocations.length > 0) {
    parts.push(
      `UNRESOLVED LOCATIONS: Weather data could NOT be retrieved for: ${context.failedLocations.join(", ")}. Clearly state in your answer that data for ${context.failedLocations.join(", ")} could not be resolved or fetched.`
    );
  }

  if (context.current) {
    const c = context.current;
    parts.push(
      `EXACT CURRENT OBSERVATIONS:
- Temperature: ${c.temperature}°C (MUST NOT be altered)
- Feels Like: ${c.feelsLike}°C
- Condition: "${c.condition}"
- Precipitation Probability: ${c.precipitationProbability}% (CRITICAL: MUST NOT be rounded to 100% or converted to certainty unless provider value is 100%)
- Rain Amount: ${c.rain || c.precipitation} mm/hr ${c.rain === 0 && c.precipitation === 0 ? "(CURRENTLY DRY: 0 mm/hr. Rain probability is for forecast window only)" : "(ACTIVE PRECIPITATION is falling now)"}
- Wind Speed: ${c.windSpeed} km/h (Gusts: ${c.windGusts} km/h)
- Humidity: ${c.humidity !== null ? `${c.humidity}%` : "Unavailable (sensor reading missing; do NOT state 0%)"}
- UV Index: ${c.uvIndex}
- Visibility: ${c.visibility !== null && c.visibility !== undefined ? `${(c.visibility / 1000).toFixed(1)} km` : "Unavailable (sensor reading missing)"}
- Pressure: ${c.pressure !== null && c.pressure !== undefined ? `${c.pressure} hPa` : "Unavailable (sensor reading missing)"}`
    );
  }

  if (context.analysis) {
    const a = context.analysis;
    parts.push(
      `DETERMINISTIC ANALYSIS: Rain Risk: ${a.rainRiskLabel}, Heat Risk: ${a.heatRiskLabel}, Wind Risk: ${a.windRiskLabel}, UV Risk: ${a.uvRiskLabel}, Visibility Risk: ${a.visibilityRiskLabel}, Comfort: ${a.comfortLabel}, Outdoor Suitability Score: ${a.outdoorSuitabilityScore}/100.`
    );
  }

  if (context.advisories && context.advisories.length > 0) {
    parts.push(`VERIFIED ADVISORIES: ${context.advisories.map((adv) => `[${adv.severity.toUpperCase()}] ${adv.title}: ${adv.description} (${adv.recommendation})`).join(" | ")}`);
  }

  if (context.activity) {
    const act = context.activity;
    parts.push(
      `ACTIVITY EVALUATION (${act.activityName}): Suitability: ${act.suitability.toUpperCase()} (Score: ${act.suitabilityScore}/100), Risk Level: ${act.riskLevel.toUpperCase()}. Reasons: ${act.primaryReasons.join("; ")}. Better window: ${act.bestTimeWindow || "None specified"}.`
    );
  }

  if (context.travel) {
    const tr = context.travel;
    parts.push(
      `TRAVEL ROUTE WEATHER (${tr.origin.name} -> ${tr.destination.name} by ${tr.mode}): Total Distance: ${tr.totalDistanceKm} km, Duration: approx ${Math.floor(
        tr.estimatedDurationMinutes / 60
      )}h ${tr.estimatedDurationMinutes % 60}m. Route Atmospheric Risk: ${tr.weatherRiskLevel.toUpperCase()}. Key Weather Observations: ${tr.keyWeatherRisks.join("; ")}.`
    );
    parts.push("IMPORTANT NOTICE: Real-time traffic, congestion, road closures, and flight delays are NOT available. Frame travel advice purely as atmospheric possibilities.");
  }

  if (context.comparison) {
    const comp = context.comparison;
    parts.push(`COMPARISON SUMMARY: ${comp.comparativeSummary}`);
    parts.push(`METRIC HIGHLIGHTS: ${comp.metricDeltas.map((m) => `${m.metric}: ${m.highlight}`).join(" | ")}`);
  }

  if (context.hourly && context.hourly.length > 0) {
    const next6 = context.hourly.slice(0, 6).map((h) => `${h.formattedTime}: ${h.temperature}°C, ${h.condition}, Rain: ${h.precipitationProbability}%`).join(" | ");
    parts.push(`NEXT 6 HOURS OUTLOOK: ${next6}`);
  }

  if (context.daily && context.daily.length > 0) {
    const next3 = context.daily.slice(0, 3).map((d) => `${d.dayName}: High ${d.tempMax}°C / Low ${d.tempMin}°C, ${d.condition}, Rain: ${d.precipitationProbabilityMax}%`).join(" | ");
    parts.push(`DAILY OUTLOOK: ${next3}`);
  }

  return parts.join("\n\n");
}

export function generateDeterministicFallback(context: GroundedWeatherContext): string {
  return buildMultilingualDeterministicFallback(context);
}

export async function transcribeAudioWithGemini(
  audioBase64: string,
  mimeType = "audio/webm",
  language?: string
): Promise<string> {
  const ai = getGenAI();
  if (!ai) {
    throw new Error("GEMINI_API_KEY is not configured");
  }

  const audioPart = {
    inlineData: {
      mimeType: mimeType || "audio/webm",
      data: audioBase64,
    },
  };

  const promptText = language
    ? `Transcribe this spoken audio accurately in ${language}. Return ONLY the transcribed text without commentary.`
    : "Transcribe this spoken audio accurately. Return ONLY the transcribed text without commentary.";

  const response = await ai.models.generateContent({
    model: "gemini-3.5-transcribe",
    contents: { parts: [audioPart, { text: promptText }] },
  });

  return (response.text || "").trim();
}
